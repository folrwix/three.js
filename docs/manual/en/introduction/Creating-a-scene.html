// Основная часть ладони let palmBase = new Box(60, 90, 8); // Прямоугольник: ширина 60 мм, длина 90 мм, высота 8 мм palmBase.setColor("lightgray"); // Закругленная крыша на ладони let roundRoof = new RoundRoof(60, 30, 6); // Ширина 60 мм, длина 30 мм, высота 6 мм roundRoof.setColor("lightgray"); // Объединение форм let palm = palmBase.combine(roundRoof); // Отверстия для ремешков let hole1 = new Box(5, 20, 8); // Отверстие: ширина 5 мм, длина 20 мм, высота 8 мм hole1.setHole(true); hole1.setPosition(10, 0, 0); // Переместить отверстие на одну сторону ладони let hole2 = hole1.copy(); // Копируем отверстие для симметрии hole2.setPosition(-10, 0, 0); // Перемещаем на другую сторону ладони // Объединение отверстий с ладонью palm = palm.combine(hole1); palm = palm.combine(hole2); // Пальцы let fingerBase = new Cylinder(10, 20); // Основание пальца: диаметр 10 мм, высота 20 мм fingerBase.setPosition(20, 20, 8); // Разместим первый палец на ладони // Дополнительные фаланги пальца let fingerMiddle = new Cylinder(10, 15); // Средняя фаланга fingerMiddle.setPosition(20, 40, 15); // Разместим выше let fingerTip = new Cylinder(10, 10); // Кончик пальца fingerTip.setPosition(20, 60, 25); // Разместим в конце // Объединяем фаланги в палец let finger = fingerBase.combine(fingerMiddle); finger = finger.combine(fingerTip); // Множество пальцев let fingers = []; for (let i = 0; i < 5; i++) { let fingerClone = finger.copy(); fingerClone.setPosition(20 + i * 15, 0, 0); // Расставим пальцы на ладони fingers.push(fingerClone); } // Собираем всю модель протеза let prostheticHand = palm; for (let i = 0; i < fingers.length; i++) { prostheticHand = prostheticHand.combine(fingers[i]); } // Экспорт модели prostheticHand.exportAsSTL();
